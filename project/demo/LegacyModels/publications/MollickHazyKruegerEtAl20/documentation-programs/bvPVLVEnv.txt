IsEpochStart() returns: bool
  LocalVars (0 vars)
  if (use_trial_gp)
    // must be both start of epoch AND start of trial_gp
    if (network.group <= 0 && network.tick <= 0)
      return true
    else
      return false
  // not using eco trials (and thus network.group) - uses the more conventional network.trial and trials_per_epoch
   instead
  else
    // CAUTION: watch to get indexing correct!!!! 
    if (network.trial <= 0 || network.trial >= trials_per_epoch)
      // just a precaution in case didn't get reset for some reason
      network.trial = 0
      return true
    else
      return false

SetScVal(int ntrials) returns: double
  LocalVars (2 vars)
    sc_val = 0 (real)
    increment = 0.1 (real)
  sc_val = 0
  sc_val = 1 - increment * (ntrials -1)
