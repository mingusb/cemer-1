cmake_minimum_required(VERSION 3.24)
project(cemer_vector_blend_tests LANGUAGES CXX)

enable_testing()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|i[3-6]86)$")
  message(FATAL_ERROR "These SIMD regression tests require an x86 processor.")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  message(FATAL_ERROR "These standalone SIMD regression tests require Clang or GCC.")
endif()
foreach(instruction_set sse2 sse4.1)
  string(REPLACE "." "_" target "vector_blend_${instruction_set}")
  add_executable(${target} vector_blend_test.cpp)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -march=x86-64 "-m${instruction_set}")
  add_test(NAME ${target} COMMAND ${target})
  set_tests_properties(${target} PROPERTIES SKIP_RETURN_CODE 77)
endforeach()
